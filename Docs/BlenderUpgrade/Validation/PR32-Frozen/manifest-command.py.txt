from pathlib import Path
import subprocess,hashlib,json
win=Path('/workspace/emberfall_win');ios=Path('/workspace/emberfall_ios');out=Path('/workspace/scratch/blender-upgrade/b3-evidence/Docs/BlenderUpgrade/Validation/PR32-Frozen')
refs={'win':'cda8a9045b3ea6e11b352cff40af8a39b5939e4c','ios':'fbd3b7011fc702b6d9921ec8138ebced1efc3041'}
def git(repo,*args):return subprocess.check_output(['git',*args],cwd=repo)
def manifest(repo,rev,folders):
 names=git(repo,'ls-tree','-r','--name-only','-z',rev,*folders).decode().split('\0')[:-1]
 return {n:hashlib.sha256(git(repo,'show',rev+':'+n)).hexdigest() for n in names}
w=manifest(win,refs['win'],['Assets/Scripts','Assets/Resources','Tests','Tools']);i=manifest(ios,refs['ios'],['Assets/Scripts','Assets/Resources','Tests','Tools'])
runtime=[p for p in w if p.startswith(('Assets/Scripts/','Assets/Resources/'))];different=[p for p in runtime if i.get(p)!=w[p]]
baseline={'win':'25096ba7dbf6e9d7ba9463ec29104c2c462da426','ios':'1cd7ce36c77064757899788e23520fb796266888'}
for path in different:
 assert w[path]==hashlib.sha256(git(win,'show',baseline['win']+':'+path)).hexdigest() and i[path]==hashlib.sha256(git(ios,'show',baseline['ios']+':'+path)).hexdigest(),path

report={'scope':'Frozen PR32 source/input identity; no E01–E06 changes included','commits':refs,'trees':{k:git(win if k=='win' else ios,'rev-parse',v+'^{tree}').decode().strip() for k,v in refs.items()},'runtimeFiles':len(runtime),'runtimeCsFiles':sum(p.endswith('.cs') for p in runtime),'preservedPlatformRuntimeDifferences':different,'newRuntimeDifferences':[],'sameNamedInputDifferences':[p for p in w.keys()&i.keys() if w[p]!=i[p]],'iosOnlyInputs':sorted(i.keys()-w.keys()),'winOnlyInputs':sorted(w.keys()-i.keys()),'winManifest':w,'iosManifest':i}
(out/'source-manifest.json').write_text(json.dumps(report,indent=2)+'\n')
print('Frozen shared runtime files byte-identical:',len(runtime),'C#:',report['runtimeCsFiles'],'input differences',len(report['sameNamedInputDifferences']))
